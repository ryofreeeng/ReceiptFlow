# Google Drive APIを呼び出す本体ライブラリ（import名: googleapiclient）
google-api-python-client

# Googleへのブラウザ認証（OAuth）を処理するライブラリ（import名: google_auth_oauthlib）
google-auth-oauthlib

# google-api-python-clientが内部で使うHTTP通信部品（直接importしない）
google-auth-httplib2

# 認証情報（トークン）の管理ライブラリ（import名: google.oauth2 / google.auth）
# google-auth-oauthlib をインストールすると自動で入るが、明示的に記載
google-auth

# .envファイルを読み込んで環境変数にセットするライブラリ
python-dotenv

# PDFを開いてページを画像データに変換するライブラリ（import名: fitz）
PyMuPDF

# 画像から文字を認識するOCRライブラリ。日本語・英語に対応（import名: easyocr）
easyocr

# 画像から文字を認識するOCRライブラリ。EasyOCRより精度が高い傾向（import名: paddleocr）
# OCR_ENGINE = "paddleocr" のときに使用する
paddleocr

# paddleocr の実行エンジン本体。paddleocr とは別に明示的なインストールが必要
paddlepaddle

# 画像データを数値配列として扱うライブラリ。PyMuPDFとEasyOCRの橋渡し役（import名: numpy）
numpy

# 画像の前処理（グレースケール変換・2値化・コントラスト強調・ノイズ除去）に使うライブラリ（import名: cv2）
opencv-python
